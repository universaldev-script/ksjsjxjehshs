    local trackBar = Instance.new("Frame", con)

    trackBar.Position = UDim2.new(0, 12, 0, 28)
    trackBar.BackgroundColor3 = THEME.Panel
    corner(trackBar, 3)

    local fill = Instance.new("Frame", trackBar)
    fill.Size = UDim2.new(math.clamp((defaultV - minV)/(maxV - minV), 0, 1), 0, 1, 0)
    fill.BackgroundColor3 = THEME.Blue
    corner(fill, 3)

    local isDragging = false
    local function updateFromPos(x)
        local rel = math.clamp((x - trackBar.AbsolutePosition.X) / trackBar.AbsoluteSize.X, 0, 1)
        fill.Size = UDim2.new(rel, 0, 1, 0)
        local val = math.floor(minV + (maxV - minV) * rel + 0.5)
        valLabel.Text = tostring(val) .. (unit or "")
        callback(val)
    end

    trackBar.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then

        end
    end))

        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            isDragging = false
        end
    end))
end

        if resizing and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            local d = (i.Position - startPos) / ViewportScale.Scale
            local newW = math.clamp(startSize.X + d.X, 520, 1100)
            local newH = math.clamp(startSize.Y + d.Y, 360, 750)
            MainFrame.Size = UDim2.fromOffset(newW, newH)
        end
    end))
    track(UserInputService.InputEnded:Connect(function(i)
        if not resizing then return end
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            resizing = false
        end
    end))
end

-- Fly Implementation
local flyBodyGyro, flyBodyVel
track(RunService.RenderStepped:Connect(function()
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not (state.fly and hrp) then
        if flyBodyGyro then flyBodyGyro:Destroy(); flyBodyGyro = nil end
        if flyBodyVel then flyBodyVel:Destroy(); flyBodyVel = nil end
        return
    end

    if not flyBodyGyro then
        flyBodyGyro = Instance.new("BodyGyro", hrp)
        flyBodyGyro.P = 9e4
        flyBodyGyro.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
        flyBodyGyro.CFrame = hrp.CFrame
    end
    if not flyBodyVel then
        flyBodyVel = Instance.new("BodyVelocity", hrp)
        flyBodyVel.Velocity = Vector3.new(0, 0, 0)
        flyBodyVel.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    end

    local cam = Workspace.CurrentCamera
    local dir = Vector3.new()
    if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + cam.CFrame.LookVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - cam.CFrame.LookVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - cam.CFrame.RightVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + cam.CFrame.RightVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0, 1, 0) end
    if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then dir = dir - Vector3.new(0, 1, 0) end

    flyBodyGyro.CFrame = cam.CFrame
    flyBodyVel.Velocity = dir * config.flySpeed
end))

-- Auto Break Eggs Loop
task.spawn(function()
    while true do
        task.wait(config.hitDelay)
        if state.autoHitEgg and remotes.EggHitRequest then
            pcall(function()
                equipPickaxe()
                local char = LocalPlayer.Character
                local hrp = char and char:FindFirstChild("HumanoidRootPart")
                if not hrp then return end

                local eggs = getBreakableEggs()
                local nearest, nearestDist = nil, config.hitRange

                for _, egg in ipairs(eggs) do
                    local dist = (egg.Position - hrp.Position).Magnitude
                    if dist <= nearestDist then
                        local eggName = egg.Parent and egg.Parent.Name or egg.Name
                        if config.targetEggs[eggName] ~= false then
                            nearest = egg
                            nearestDist = dist
                        end
                    end
                end

                if nearest then
                    remotes.EggHitRequest:FireServer(nearest)
                end
            end)
        end
    end
end)

                if not hrp then return end

                -- Find prompts
                local targetPrompt = nil
                for _, p in ipairs(Workspace:GetDescendants()) do
                    if p:IsA("ProximityPrompt") and p.Name == "StealPrompt" and p.Enabled then
                        targetPrompt = p
                        break
                    end
                end

                if targetPrompt and targetPrompt.Parent then
                    local part = targetPrompt.Parent:IsA("BasePart") and targetPrompt.Parent or targetPrompt.Parent:FindFirstChildWhichIsA("BasePart")
                    if part then
                        -- Teleport close, fire prompt, and instant bank
                        char:PivotTo(part.CFrame * CFrame.new(0, 2, 0))
                        task.wait(0.1)
                        fireproximityprompt(targetPrompt, 0)
                        task.wait(0.1)

                        if state.instantBank then
                            local hb = getMyPlotHitbox()
                            if hb then
                                char:PivotTo(hb.CFrame * CFrame.new(0, 3, 0))
                            end
                        end
                    end
                end
            end)
        end
    end
end)
